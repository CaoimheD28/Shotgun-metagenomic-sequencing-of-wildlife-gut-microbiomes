#!/bin/bash -l
#SBATCH --job-name=plasmid_assembly_SAMPLE
#SBATCH -N 1
#SBATCH --cpus-per-task=8
#SBATCH -t 4:00:00

# Activate environment
conda activate abricate_env

# Path to your assembly file
ASSEMBLY="/PATH/TO/ASSEMBLY.fasta"

# Output directory
OUTDIR="/PATH/TO/OUTPUT_FOLDER"
mkdir -p "$OUTDIR"

# Base name for output
BASENAME=$(basename "$ASSEMBLY" .fasta)


# Run ABRicate with PlasmidFinder
abricate --db plasmidfinder "$ASSEMBLY" > "$OUTDIR/${BASENAME}_plasmidfinder.tsv"
