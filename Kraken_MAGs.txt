#!/bin/bash -l
#SBATCH --job-name=kraken2_SAMPLE_mags
#SBATCH --output=kraken2_SAMPLE_%j.out
#SBATCH --error=kraken2_SAMPLE_%j.err
#SBATCH -N 1
#SBATCH --cpus-per-task=4
#SBATCH -t 99:00:00

# Activate Kraken2 environment
conda activate kraken2_env

# Run Kraken2 on each .fa file in the metabat2/bins directory
for fa_file in /PATH/TO/BINS/*.fa; do
    sample_name=$(basename "$fa_file" .fa)

    kraken2 \
        --db /PATH/TO/kraken2_db/k2_standard_08gb_20240112 \
        --output "/PATH/TO/METABAT2_OUTPUT_FOLDER/kraken2_output/${sample_name}_kraken2_output.txt" \
        --report "/PATH/TO/METABAT2_OUTPUT_FOLDER/kraken2_output/${sample_name}_kraken2_report.txt" \
        --threads 4 \
        "$fa_file"
